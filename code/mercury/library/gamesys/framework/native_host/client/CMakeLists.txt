# Independent consumer: configure this directory after installing NativeHost.
cmake_minimum_required(VERSION 3.24)
project(NativeHostClient VERSION 0.1.0 LANGUAGES CXX OBJCXX)
find_package(NativeHost 0.1 CONFIG REQUIRED)
find_library(CLIENT_COCOA Cocoa REQUIRED)
find_library(CLIENT_METAL Metal REQUIRED)
find_library(CLIENT_QUARTZCORE QuartzCore REQUIRED)
find_package(Threads REQUIRED)
add_executable(native_host_demo demo.mm)
target_link_libraries(native_host_demo PRIVATE NativeHost::Window "${CLIENT_METAL}" "${CLIENT_QUARTZCORE}" "${CLIENT_COCOA}" Threads::Threads)
add_executable(native_host_conformance conformance.mm)
target_link_libraries(native_host_conformance PRIVATE NativeHost::Window "${CLIENT_QUARTZCORE}" "${CLIENT_COCOA}" Threads::Threads)
foreach(target native_host_demo native_host_conformance)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -fobjc-arc -Wall -Wextra)
    set_target_properties(${target} PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME native_host_conformance COMMAND native_host_conformance)
    set_tests_properties(native_host_conformance PROPERTIES LABELS "hardware;window" TIMEOUT 30)
    add_test(NAME native_host_demo COMMAND native_host_demo --test)
    set_tests_properties(native_host_demo PROPERTIES LABELS "hardware;window" TIMEOUT 40)
endif()
